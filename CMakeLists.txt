cmake_minimum_required(VERSION 3.20)
project(FilesystemAPFS VERSION 1.0.0 LANGUAGES C)
include(CTest)
find_package(OpenSSL REQUIRED)
include(FetchContent)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(INFILTRATR_COMMON_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(INFILTRATR_COMMON_BUILD_SHARED OFF CACHE BOOL "" FORCE)
FetchContent_Declare(infiltratr_common GIT_REPOSITORY https://github.com/Infiltrator-Projects/Libraries.git GIT_TAG c76ec95ef66fc728df6a906ad75a6a4dd63b419b)
FetchContent_MakeAvailable(infiltratr_common)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.h" "#define LD_VERSION \"1.0.0\"\n")
add_library(apfs-driver SHARED native/apfs_native.c native/apfs_worker.c module/apfs_module.c compat/legacy_services.c
 fixtures/apfs_codec.c compat/fixture_bridge.c)
set_source_files_properties(native/apfs_worker.c PROPERTIES COMPILE_DEFINITIONS "main=ifs_apfs_worker_main")
set_target_properties(apfs-driver PROPERTIES PREFIX "" OUTPUT_NAME "APFS" SUFFIX ".so")
target_include_directories(apfs-driver PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/native" "${CMAKE_CURRENT_SOURCE_DIR}/compat" "${CMAKE_CURRENT_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/include" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(apfs-driver PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_compile_options(apfs-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(apfs-driver PRIVATE InfiltratrCommon::Common OpenSSL::Crypto)
if(BUILD_TESTING)
 add_executable(apfs-module-test tests/test_apfs_module.c)
 target_include_directories(apfs-module-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
 target_link_libraries(apfs-module-test PRIVATE apfs-driver)
 add_test(NAME apfs-module-interface COMMAND apfs-module-test)
    add_executable(apfs-original-native-test tests/test_apfs_native.c)
    target_include_directories(apfs-original-native-test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/native")
    target_compile_definitions(apfs-original-native-test PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
    target_link_libraries(apfs-original-native-test PRIVATE apfs-driver OpenSSL::Crypto)
    add_test(NAME apfs-original-native COMMAND apfs-original-native-test)
endif()
install(TARGETS apfs-driver LIBRARY DESTINATION /System/Filesystems)
# Install only this filesystem's illustrated PDF beside its native shared object.
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/docs/APFS.pdf"
        DESTINATION /System/Resources/Documentation/Filesystem-APFS)
